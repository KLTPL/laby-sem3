USE aparaty;

CREATE TABLE Producent (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    nazwa VARCHAR(50) NOT NULL,
    kraj VARCHAR(20) NOT NULL DEFAULT 'nieznany',
    adresKorespondencyjny VARCHAR(100) NOT NULL DEFAULT 'nieznany',
    CONSTRAINT producent_nazwa_niepusta CHECK (TRIM(nazwa) <> '')
);

CREATE TABLE Matryca (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    przekatna DECIMAL(4, 2),
    rozdzielczosc DECIMAL(3, 1),
    typ VARCHAR(10),
    CONSTRAINT matryca_przekatna_nieujemna CHECK (przekatna >= 0),
    CONSTRAINT matryca_rozdzielczosc_nieujemna CHECK (rozdzielczosc >= 0)
) AUTO_INCREMENT = 100;

CREATE TABLE Obiektyw (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    model VARCHAR(30),
    minPrzeslona FLOAT,
    maxPrzeslona FLOAT,
    CONSTRAINT obiektyw_min_nieujemna CHECK (minPrzeslona >= 0),
    CONSTRAINT obiektyw_max_nieujemna CHECK (maxPrzeslona >= 0),
    CONSTRAINT obiektyw_min_mniejsza CHECK (minPrzeslona < maxPrzeslona)
);

CREATE TABLE Aparat (
    model VARCHAR(30) PRIMARY KEY,
    producent INT,
    matryca INT,
    obiektyw INT,
    waga FLOAT,
    typ ENUM('kompaktowy', 'lustrzanka', 'profesjonalny', 'inny'),
    CONSTRAINT aparat_waga_nieujemna CHECK (waga >= 0),
    CONSTRAINT aparat_producent_fk FOREIGN KEY (producent)
    REFERENCES Producent (ID),
    CONSTRAINT aparat_matryca_fk FOREIGN KEY (matryca)
    REFERENCES Matryca (ID),
    CONSTRAINT aparat_obiektyw_fk FOREIGN KEY (obiektyw)
    REFERENCES Obiektyw (ID)
);

SHOW TABLES;

-- Tworzy root (289936 nie ma CREATE ROUTINE).
DROP PROCEDURE IF EXISTS generuj_aparaty;

DELIMITER //

CREATE PROCEDURE generuj_aparaty()
BEGIN
    DECLARE nr INT;
    DECLARE koniec INT;

    SELECT COALESCE(MAX(CAST(SUBSTRING(model, 8) AS UNSIGNED)), 0)
    INTO nr
    FROM Aparat
    WHERE model REGEXP '^Aparat-[0-9]+$';

    SET koniec = nr + 100;

    WHILE nr < koniec DO
        SET nr = nr + 1;
        INSERT INTO Aparat (model, producent, matryca, obiektyw, waga, typ)
        VALUES (
            CONCAT('Aparat-', nr),
            (SELECT ID FROM Producent ORDER BY RAND() LIMIT 1),
            (SELECT ID FROM Matryca ORDER BY RAND() LIMIT 1),
            (SELECT ID FROM Obiektyw ORDER BY RAND() LIMIT 1),
            ROUND(150 + RAND() * 1350),
            ELT(1 + FLOOR(RAND() * 4),
                'kompaktowy', 'lustrzanka', 'profesjonalny', 'inny')
        );
    END WHILE;
END //

DELIMITER ;

DROP FUNCTION IF EXISTS najmniejsza_matryca;

DELIMITER //

CREATE FUNCTION najmniejsza_matryca(p_producent INT)
RETURNS VARCHAR(30)
NOT DETERMINISTIC
READS SQL DATA
BEGIN
    RETURN (
        SELECT a.model
        FROM Aparat AS a
        INNER JOIN Matryca AS m ON a.matryca = m.ID
        WHERE a.producent = p_producent
        ORDER BY m.przekatna ASC, a.model ASC
        LIMIT 1
    );
END //

DELIMITER ;

SELECT najmniejsza_matryca(1) AS canon, najmniejsza_matryca(999) AS brak;

-- Odpowiedź:
-- DETERMINISTIC to tylko deklaracja: MySQL nie sprawdza, czy jest prawdziwa.
-- Oznacza że dane argumenty mają zawsze dawać ten sam wynik. Nasza funkcja czyta
-- tabele, więc wynik zmienia się po zmianie danych a więc nie jest deterministyczna.
-- Konsekwencje oznaczenia jej DETERMINISTIC:
-- - SQL traktuje wywołanie ze stałym argumentem jak stałą więc liczy je
--   raz i może użyć indeksu
-- - przy fałszywej deklaracji wynik zapytania może być błędny, gdy dane
--   zmieniają się w trakcie (np. w przez wywołanie UPDATE;

DROP FUNCTION IF EXISTS sprawdz_aparat;

DELIMITER //

CREATE FUNCTION sprawdz_aparat(
    id_producenta INT,
    id_matrycy INT,
    id_obiektywu INT
)
RETURNS BOOLEAN
READS SQL DATA
BEGIN
    RETURN EXISTS (SELECT 1 FROM Producent WHERE ID = id_producenta)
        AND EXISTS (SELECT 1 FROM Matryca WHERE ID = id_matrycy)
        AND EXISTS (SELECT 1 FROM Obiektyw WHERE ID = id_obiektywu);
END //

DELIMITER ;

-- 1 = TRUE, 0 = FALSE.
SELECT
    sprawdz_aparat(1, 100, 1) AS wszystkie,
    sprawdz_aparat(999, 100, 1) AS brak_producenta,
    sprawdz_aparat(1, 999, 1) AS brak_matrycy,
    sprawdz_aparat(1, 100, 999) AS brak_obiektywu;

DROP TRIGGER IF EXISTS blokuj_zmiane_obiektywu;
DROP TRIGGER IF EXISTS blokuj_usuniecie_obiektywu;

DELIMITER //

-- dwa wyzwalacze dla dwóch zdarzeń
CREATE TRIGGER blokuj_zmiane_obiektywu
BEFORE UPDATE ON Obiektyw
FOR EACH ROW
BEGIN
    DECLARE v_komunikat VARCHAR(128);

    IF EXISTS (SELECT 1 FROM Aparat WHERE obiektyw = OLD.ID) THEN
        SET v_komunikat = CONCAT(
            'Nie można zmienić obiektywu ', OLD.ID,
            ': jest przypisany do aparatu'
        );
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = v_komunikat;
    END IF;
END //

CREATE TRIGGER blokuj_usuniecie_obiektywu
BEFORE DELETE ON Obiektyw
FOR EACH ROW
BEGIN
    DECLARE v_komunikat VARCHAR(128);

    IF EXISTS (SELECT 1 FROM Aparat WHERE obiektyw = OLD.ID) THEN
        SET v_komunikat = CONCAT(
            'Nie można usunąć obiektywu ', OLD.ID,
            ': jest przypisany do aparatu'
        );
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = v_komunikat;
    END IF;
END //

DELIMITER ;

CREATE TABLE HistoriaProducentow (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_producenta INT NOT NULL,
    stary_kraj VARCHAR(20),
    nowy_kraj VARCHAR(20),
    data_zmiany DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT historia_producent_fk FOREIGN KEY (id_producenta)
    REFERENCES Producent (ID) ON DELETE CASCADE
);

DROP TRIGGER IF EXISTS zapisz_zmiane_kraju;

DELIMITER //

CREATE TRIGGER zapisz_zmiane_kraju
AFTER UPDATE ON Producent
FOR EACH ROW
BEGIN
    IF NOT (OLD.kraj <=> NEW.kraj) THEN
        INSERT INTO HistoriaProducentow (id_producenta, stary_kraj, nowy_kraj)
        VALUES (OLD.ID, OLD.kraj, NEW.kraj);
    END IF;
END //

DELIMITER ;
